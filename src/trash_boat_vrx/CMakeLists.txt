cmake_minimum_required(VERSION 3.16)
project(trash_boat_vrx)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-transport13 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
find_package(gz-msgs10 REQUIRED)
add_library(TrashWorld SHARED src/TrashWorld.cc)
target_compile_features(TrashWorld PUBLIC cxx_std_17)
target_compile_options(TrashWorld PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(TrashWorld PRIVATE include)
target_link_libraries(TrashWorld gz-sim8::core gz-transport13::core
  gz-plugin2::register gz-msgs10::gz-msgs10)
install(TARGETS TrashWorld DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/generate_world scripts/water_detector DESTINATION lib/${PROJECT_NAME})
ament_environment_hooks(hooks/paths.dsv.in hooks/paths.sh)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(test_intake test/test_intake.cpp)
  target_include_directories(test_intake PRIVATE include)
  target_link_libraries(test_intake gz-sim8::core)
  ament_add_pytest_test(test_python test TIMEOUT 90)
endif()
ament_package()
